// ================================================================
// MATHEMATICAL STATE
// ================================================================
//
// Normalized race coordinate:
//
// Achilles initially = 0
// Tortoise initially = 0.5
// Meeting point      = 1.0
//
// Remaining distance to meeting point:
//
// Stage 1 = 1/2
// Stage 2 = 1/4
// Stage 3 = 1/8
// ...
// Stage n = 1/2^n
//
// Position at stage n:
//
// x = 1 - 1/2^n
// ================================================================

double currentRem = 1.0;

double targetRem = 0.5;

double animStartRem = 1.0;

double animEndRem = 0.5;

int stage = 1;

final int MAX_STAGE = 122;


// ================================================================
// ANIMATION
// ================================================================

boolean autoPlay = false;

// ================================================================
// ANIMATION
// ================================================================

void updateAnimation() {

  if (!moving && !autoPlay) {

    return;
  }


  // ==============================================================
  // AUTO DIVE
  // ==============================================================

  if (autoPlay) {

    moving = true;


    // ------------------------------------------------------------
    // Exponential approach
    // ------------------------------------------------------------

    currentRem *= 0.95;


    // ------------------------------------------------------------
    // Prevent numerical underflow
    // ------------------------------------------------------------

    if (currentRem < Math.pow(2.0, -MAX_STAGE)) {

      currentRem =
        Math.pow(2.0, -MAX_STAGE);
    }


    // ------------------------------------------------------------
    // Determine Zeno stage
    // ------------------------------------------------------------

    int newStage =
      (int)Math.floor(
        -Math.log(currentRem) /
        Math.log(2.0)
      ) + 1;


    if (
      newStage > stage &&
      newStage <= MAX_STAGE
    ) {

      stage = newStage;

      flashAlpha = 40;
    }


    // ------------------------------------------------------------
    // FINAL STAGE
    // ------------------------------------------------------------

    if (currentRem <=
        Math.pow(2.0, -MAX_STAGE)) {

      stage = MAX_STAGE;

      currentRem =
        Math.pow(2.0, -MAX_STAGE);

      targetRem =
        currentRem;

      autoPlay = false;

      moving = false;

      flashAlpha = 150;

      return;
    }


    // ------------------------------------------------------------
    // Current target
    // ------------------------------------------------------------

    targetRem =
      Math.pow(
        2.0,
        -stage
      );


    animStartRem = currentRem;

    animEndRem = targetRem;

    return;
  }


  // ==============================================================
  // MANUAL ZENO STEP
  // ==============================================================

  animationFrame++;


  float t =
    constrain(
      (float)animationFrame /
      (float)ANIMATION_FRAMES,

      0.0,

      1.0
    );


  // --------------------------------------------------------------
  // Smoothstep interpolation
  // --------------------------------------------------------------

  float smooth =
    t *
    t *
    (3.0 - 2.0 * t);


  currentRem =
    animStartRem +
    (
      animEndRem -
      animStartRem
    ) *
    smooth;


  // --------------------------------------------------------------
  // STEP COMPLETE
  // --------------------------------------------------------------

  if (t >= 1.0) {

    currentRem =
      animEndRem;

    moving = false;

    flashAlpha = 100;


    if (stage < MAX_STAGE) {

      stage++;

      targetRem =
        Math.pow(
          2.0,
          -stage
        );
    }
  }
}


// ================================================================
// THRESHOLD-ANCHORED CAMERA
// ================================================================

void updateCamera() {

  dynamicZoomFactor +=
    (
      deepZoomMode
      ?
      1.0 - dynamicZoomFactor
      :
      0.0 - dynamicZoomFactor
    ) *
    0.1;


  // --------------------------------------------------------------
  // ZOOM THRESHOLD
  // --------------------------------------------------------------

  double zoomAnchorThreshold = 0.15;

  double idealZoom = 1.0;


  if (
    currentRem <
    zoomAnchorThreshold
  ) {

    idealZoom =
      zoomAnchorThreshold /
      currentRem;
  }


  // --------------------------------------------------------------
  // Smooth camera
  // --------------------------------------------------------------

  currentZoom =
    1.0 +
    (
      idealZoom - 1.0
    ) *
    dynamicZoomFactor;
}


// ================================================================
// MATHEMATICS → SCREEN
// ================================================================

float mapToScreen(
  double remainingMath
) {

  double scaledRem =
    remainingMath *
    currentZoom;


  float widthPx =
    lineRight -
    lineLeft;


  // --------------------------------------------------------------
  // Meeting point remains fixed at the right.
  // --------------------------------------------------------------

  return (float)(
    lineRight -
    (
      scaledRem *
      widthPx
    )
  );
}


// ================================================================
// START NEXT ZENO STEP
// ================================================================

void startNextStep() {

  if (
    moving ||
    stage >= MAX_STAGE
  ) {

    return;
  }


  animStartRem =
    currentRem;


  animEndRem =
    Math.pow(
      2.0,
      -stage
    );


  animationFrame = 0;

  moving = true;
}


// ================================================================
// RESET
// ================================================================

void reset() {

  currentRem = 1.0;

  targetRem = 0.5;

  animStartRem = 1.0;

  animEndRem = 0.5;

  stage = 1;

  moving = false;

  autoPlay = false;

  dynamicZoomFactor = 1.0;

  currentZoom = 1.0;

  flashAlpha = 255;
}

// ================================================================
// ZENO BOUNDARIES
// ================================================================

void drawZenoLines() {

  float verticalTop =
    height * 0.40;

  float verticalBottom =
    height * 0.70;


  // ==============================================================
  // 122 ZENO BOUNDARIES
  // ==============================================================

  for (
    int n = 1;
    n <= MAX_STAGE;
    n++
  ) {

    // ------------------------------------------------------------
    // Remaining distance represented by this boundary
    // ------------------------------------------------------------

    double lineRem =
      Math.pow(
        2.0,
        -n
      );


    float sx =
      mapToScreen(
        lineRem
      );


    // ------------------------------------------------------------
    // Outside screen
    // ------------------------------------------------------------

    if (
      sx < -100 ||
      sx > width + 100
    ) {

      continue;
    }


    // ------------------------------------------------------------
    // Distance to neighboring boundary
    // ------------------------------------------------------------

    float gapPx =
      (float)(
        Math.abs(
          mapToScreen(lineRem) -
          mapToScreen(lineRem * 2.0)
        )
      );


    // ------------------------------------------------------------
    // Very small subpixel boundary
    // ------------------------------------------------------------

    if (gapPx < 0.1) {

      continue;
    }


    // ------------------------------------------------------------
    // Transparency
    // ------------------------------------------------------------

    float lineAlpha =
      constrain(
        map(
          gapPx,

          0.2,
          15.0,

          5,
          255
        ),

        0,
        255
      );


    // ------------------------------------------------------------
    // Has Achilles passed this boundary?
    // ------------------------------------------------------------

    boolean passed =
      currentRem <=
      lineRem + 1e-18;


    // ------------------------------------------------------------
    // Current target
    // ------------------------------------------------------------

    boolean isTarget =
      (n == stage);


    int activeColor =
      C_GRID;

    int weight = 2;


    if (isTarget) {

      activeColor =
        C_STAGE;

      weight = 5;

    } else if (passed) {

      activeColor =
        C_PASSED;

      weight = 3;
    }


    // ------------------------------------------------------------
    // Boundary
    // ------------------------------------------------------------

    stroke(
      red(activeColor),
      green(activeColor),
      blue(activeColor),
      lineAlpha
    );


    strokeWeight(weight);


    // ------------------------------------------------------------
    // FULL VERTICAL BOUNDARY
    // ------------------------------------------------------------

    line(
      sx,
      verticalTop,
      sx,
      verticalBottom
    );


    // ============================================================
    // LABELS
    // ============================================================

    float textAlpha = 0;


    if (deepZoomMode) {

      if (
        n >= stage - 1 &&
        n <= stage + 3
      ) {

        textAlpha =
          255 -
          max(
            0,
            (n - stage) * 50
          );
      }

    } else {

      textAlpha =
        map(
          gapPx,

          10,
          45,

          0,
          255
        );
    }


    textAlpha =
      constrain(
        textAlpha,

        0,
        255
      );


    if (textAlpha > 5) {

      if (isTarget) {

        fill(
          red(C_STAGE),
          green(C_STAGE),
          blue(C_STAGE),
          textAlpha
        );

      } else if (passed) {

        fill(
          red(C_PASSED),
          green(C_PASSED),
          blue(C_PASSED),
          textAlpha
        );

      } else {

        fill(
          red(C_TEXT),
          green(C_TEXT),
          blue(C_TEXT),
          textAlpha
        );
      }


      noStroke();


      textSize(
        (
          n <= 6 ||
          deepZoomMode
        )
        ?
        16
        :
        12
      );


      text(
        fractionText(n),

        sx,

        verticalTop - 20
      );
    }
  }


  // ==============================================================
  // MEETING POINT WALL
  // ==============================================================

  float destX =
    lineRight;


  float pulse =
    (
      sin(
        frameCount *
        0.08f
      ) +
      1.0f
    ) *
    0.5f;


  // --------------------------------------------------------------
  // Glow
  // --------------------------------------------------------------

  stroke(
    red(C_DESTINATION),
    green(C_DESTINATION),
    blue(C_DESTINATION),
    30 + 50 * pulse
  );


  strokeWeight(12);


  line(
    destX,
    verticalTop - 25,

    destX,
    verticalBottom + 25
  );


  // --------------------------------------------------------------
  // Main wall
  // --------------------------------------------------------------

  stroke(C_DESTINATION);

  strokeWeight(7);


  line(
    destX,
    verticalTop - 25,

    destX,
    verticalBottom + 25
  );


  // --------------------------------------------------------------
  // Meeting point label
  // --------------------------------------------------------------

  fill(C_DESTINATION);

  noStroke();

  textSize(20);


  text(
    EMOJI_TARGET +
    " MEETING POINT",

    destX,

    verticalTop - 70
  );


  // --------------------------------------------------------------
  // Tap instruction
  // --------------------------------------------------------------

  if (!autoPlay) {

    fill(
      red(C_DESTINATION),
      green(C_DESTINATION),
      blue(C_DESTINATION),

      150 +
      105 * pulse
    );


    textSize(14);


    text(
      EMOJI_TAP +
      " TAP TO AUTO-DIVE",

      destX - 5,

      verticalTop - 45
    );
  }
}


// ================================================================
// ACHILLES + TORTOISE
// ================================================================

void drawRacePositions() {

  // --------------------------------------------------------------
  // Achilles position
  // --------------------------------------------------------------

  float achillesX =
    mapToScreen(
      currentRem
    );


  // --------------------------------------------------------------
  // Tortoise position
  //
  // Tortoise remains one Zeno step ahead.
  // --------------------------------------------------------------

  double tortoiseRem =
    currentRem *
    0.5;


  float tortoiseX =
    mapToScreen(
      tortoiseRem
    );


  // ==============================================================
  // TORTOISE TRACK
  // ==============================================================

  stroke(
    red(C_TORTOISE),
    green(C_TORTOISE),
    blue(C_TORTOISE),
    110
  );


  strokeWeight(3);


  line(
    lineLeft,
    lineY + 55,

    lineRight,
    lineY + 55
  );


  // ==============================================================
  // ACHILLES
  // ==============================================================

  if (
    achillesX >= lineLeft &&
    achillesX <= lineRight + 50
  ) {

    drawAchilles(
      achillesX,
      lineY - 65
    );
  }


  // ==============================================================
  // TORTOISE
  // ==============================================================

  if (
    tortoiseX >= lineLeft &&
    tortoiseX <= lineRight + 50
  ) {

    drawTortoise(
      tortoiseX,
      lineY + 55
    );
  }


  // ==============================================================
  // GAP INDICATOR
  // ==============================================================

  if (
    tortoiseX >
    achillesX
  ) {

    float gap =
      tortoiseX -
      achillesX;


    if (gap > 4) {

      stroke(C_REMAIN);

      strokeWeight(2);


      line(
        achillesX,
        lineY + 5,

        tortoiseX,
        lineY + 5
      );


      // ----------------------------------------------------------
      // Gap label
      // ----------------------------------------------------------

      fill(C_REMAIN);

      noStroke();

      textSize(13);


      text(
        EMOJI_GAP +
        " remaining chase gap",

        (
          achillesX +
          tortoiseX
        ) / 2,

        lineY - 2
      );
    }
  }
}

// ================================================================
// POSITION FORMAT
// ================================================================

String formatPosition(
  double rem
) {

  if (
    rem >= 0.000001 &&
    rem <= 1.0
  ) {

    return nf(
      (float)(
        1.0 - rem
      ),

      1,

      10
    );

  } else {

    return
      "1.0 - (" +
      scientificOrDecimal(rem) +
      ")";
  }
}


// ================================================================
// SCIENTIFIC FORMAT
// ================================================================

String scientificOrDecimal(
  double value
) {

  if (value == 0.0) {

    return "0";
  }


  double absValue =
    Math.abs(value);


  if (
    absValue >= 0.000001 &&
    absValue < 1000000.0
  ) {

    return nf(
      (float)value,

      1,

      10
    );
  }


  int exponent =
    (int)Math.floor(
      Math.log10(absValue)
    );


  double mantissa =
    value /
    Math.pow(
      10.0,
      exponent
    );


  return
    nf(
      (float)mantissa,

      1,

      6
    ) +

    " × 10^" +

    exponent;
}


// ================================================================
// FRACTION LABEL
// ================================================================

String fractionText(
  int n
) {

  // --------------------------------------------------------------
  // Exact integer fractions are safe through n = 60
  // in a signed Java long.
  // --------------------------------------------------------------

  if (n <= 60) {

    long denominator =
      1L << n;


    long numerator =
      denominator - 1L;


    return
      numerator +
      "/" +
      denominator;
  }


  // --------------------------------------------------------------
  // For deep stages avoid integer overflow.
  // --------------------------------------------------------------

  return
    "1 - (1/2^" +
    n +
    ")";
}

      width * 0.022
    )
  );


  // ==============================================================
  // RESET
  // ==============================================================

  fill(C_PANEL);

  stroke(C_GRID);

  strokeWeight(2);


  rect(
    x1,
    btnY,

    btnW,
    btnH,

    15
  );


  fill(C_TEXT);


  text(
    EMOJI_RESET +
    " RESET",

    x1 + btnW / 2,

    btnY + btnH / 2
  );


  // ==============================================================
  // PIERCE NEXT
  // ==============================================================

  fill(C_PANEL);

  stroke(C_GRID);

  strokeWeight(2);


  rect(
    x2,
    btnY,

    btnW,
    btnH,

    15
  );


  fill(C_STAGE);


  text(
    EMOJI_NEXT +
    " PIERCE NEXT",

    x2 + btnW / 2,

    btnY + btnH / 2
  );


  // ==============================================================
  // AUTO DIVE
  // ==============================================================

  fill(C_PANEL);


  stroke(
    autoPlay
    ?
    C_PASSED
    :
    C_GRID
  );


  strokeWeight(2);


  rect(
    x3,
    btnY,

    btnW,
    btnH,

    15
  );


  fill(
    autoPlay
    ?
    C_PASSED
    :
    C_TEXT
  );


  text(
    autoPlay
    ?
    EMOJI_STOP + " HALT DIVE"
    :
    EMOJI_PLAY + " AUTO DIVE",

    x3 + btnW / 2,

    btnY + btnH / 2
  );


  // ==============================================================
  // CAMERA
  // ==============================================================

  fill(C_PANEL);


  stroke(
    deepZoomMode
    ?
    C_REMAIN
    :
    C_GRID
  );


  strokeWeight(2);


  rect(
    x4,
    btnY,

    btnW,
    btnH,

    15
  );


  fill(
    deepZoomMode
    ?
    C_REMAIN
    :
    C_TEXT
  );


  text(
    deepZoomMode
    ?
    EMOJI_DYNAMIC + " DYNAMIC"
    :
    EMOJI_STATIC + " STATIC",

    x4 + btnW / 2,

    btnY + btnH / 2
  );
}


// ================================================================
// TOUCH CONTROL
// ================================================================

void mousePressed() {

  // ==============================================================
  // MEETING POINT TAP
  // ==============================================================

  if (
    mouseX >
    lineRight - 150 &&

    mouseY >
    height * 0.25 &&

    mouseY <
    height * 0.75
  ) {

    if (!autoPlay) {

      autoPlay = true;

      deepZoomMode = true;

    } else {

      autoPlay = false;
    }


    return;
  }


  // ==============================================================
  // BUTTON GEOMETRY
  // ==============================================================

  float btnY =
    height * 0.82;


  float btnH =
    height * 0.12;


  float gap =
    width * 0.04;


  float btnW =
    width * 0.20;


  float x1 =
    gap;


  float x2 =
    x1 +
    btnW +
    gap;


  float x3 =
    x2 +
    btnW +
    gap;


  float x4 =
    x3 +
    btnW +
    gap;


  // ==============================================================
  // BUTTON AREA
  // ==============================================================

  if (
    mouseY > btnY &&
    mouseY < btnY + btnH
  ) {


    // ============================================================
    // RESET
    // ============================================================

    if (
      mouseX > x1 &&
      mouseX < x1 + btnW
    ) {

      reset();

      return;
    }


    // ============================================================
    // PIERCE NEXT
    // ============================================================

    if (
      mouseX > x2 &&
      mouseX < x2 + btnW
    ) {

      autoPlay = false;

      startNextStep();

      return;
    }


    // ============================================================
    // AUTO DIVE
    // ============================================================

    if (
      mouseX > x3 &&
      mouseX < x3 + btnW
    ) {

      autoPlay =
        !autoPlay;


      if (autoPlay) {

        deepZoomMode = true;
      }


      return;
    }


    // ============================================================
    // CAMERA
    // ============================================================

    if (
      mouseX > x4 &&
      mouseX < x4 + btnW
    ) {

      deepZoomMode =
        !deepZoomMode;

      return;
    }
  }


  // ==============================================================
  // ANYWHERE ELSE = NEXT ZENO STEP
  // ==============================================================

  autoPlay = false;

  startNextStep();
}
